let currentUser = null;
let cart = [];

// Cargar catálogo inicial
document.addEventListener('DOMContentLoaded', () => {
    renderProducts(productos);
    logSQL("SELECT p.*, c.nombre AS categoria, m.nombre AS marca FROM productos p JOIN categorias c ON p.categoria_id = c.id JOIN marcas m ON p.marca_id = m.id;");
});

// Renderizar Productos
function renderProducts(items) {
    const catalog = document.getElementById('catalog');
    catalog.innerHTML = '';

    if (items.length === 0) {
        catalog.innerHTML = '<p>No se encontraron productos para los filtros seleccionados.</p>';
        return;
    }

    items.forEach(p => {
        catalog.innerHTML += `
            <div class="card">
                <img src="${p.img}" alt="${p.nombre}">
                <div class="card-body">
                    <span class="badge">${p.marca} | ${p.categoria}</span>
                    <h4 class="card-title">${p.nombre}</h4>
                    <div class="price">$${p.precio.toFixed(2)}</div>
                    <button class="btn-add" onclick="addToCart(${p.id})">Agregar al Carrito</button>
                </div>
            </div>
        `;
    });
}

// Filtro por Marca y Categoría
function filterProducts() {
    const cat = document.getElementById('select-category').value;
    const brand = document.getElementById('select-brand').value;

    let filtered = productos.filter(p => {
        const matchCat = (cat === 'all' || p.categoria === cat);
        const matchBrand = (brand === 'all' || p.marca === brand);
        return matchCat && matchBrand;
    });

    renderProducts(filtered);

    // Muestra la consulta SQL correspondiente al filtro ejecutado
    logSQL(`SELECT * FROM productos WHERE ('${cat}' = 'all' OR categoria = '${cat}') AND ('${brand}' = 'all' OR marca = '${brand}');`);
}

// Registro e Inicio de Sesión
function handleRegister(e) {
    e.preventDefault();
    const name = document.getElementById('reg-name').value;
    const email = document.getElementById('reg-email').value;
    const pass = document.getElementById('reg-pass').value;

    currentUser = { id: 101, name, email };
    document.getElementById('user-display').innerText = `Hola, ${name}`;
    toggleAuthModal();

    logSQL(`INSERT INTO usuarios (nombre, email, password) VALUES ('${name}', '${email}', '${pass}');`);
    alert('Usuario registrado exitosamente en la base de datos.');
}

function handleLogin(e) {
    e.preventDefault();
    const email = document.getElementById('login-email').value;
    currentUser = { id: 101, name: email.split('@')[0], email };
    document.getElementById('user-display').innerText = `Hola, ${currentUser.name}`;
    toggleAuthModal();

    logSQL(`SELECT * FROM usuarios WHERE email = '${email}' AND password = '***';`);
    alert('Sesión iniciada correctamente.');
}

// Carrito de compras
function addToCart(id) {
    const prod = productos.find(p => p.id === id);
    cart.push(prod);
    document.getElementById('cart-count').innerText = cart.length;
    alert(`"${prod.nombre}" agregado al carrito.`);
}

function toggleCart() {
    const modal = document.getElementById('cart-modal');
    modal.style.display = (modal.style.display === 'flex') ? 'none' : 'flex';

    const cartContainer = document.getElementById('cart-items');
    cartContainer.innerHTML = '';
    let total = 0;

    cart.forEach(item => {
        total += item.precio;
        cartContainer.innerHTML += `<p>• ${item.nombre} - <strong>$${item.precio.toFixed(2)}</strong></p>`;
    });

    document.getElementById('cart-total').innerText = total.toFixed(2);
}

// Checkout / Formulario de Compra conectado a BD
function handleCheckout(e) {
    e.preventDefault();
    if (!currentUser) {
        alert('Debes iniciar sesión para realizar una compra.');
        toggleCart();
        toggleAuthModal();
        return;
    }

    if (cart.length === 0) {
        alert('El carrito está vacío.');
        return;
    }

    const address = document.getElementById('ship-address').value;
    const payment = document.getElementById('pay-method').value;
    const total = document.getElementById('cart-total').innerText;

    // Generación del comando SQL INSERT para registrar la venta
    logSQL(`INSERT INTO ventas (usuario_id, total, direccion_envio, metodo_pago) VALUES (${currentUser.id}, ${total}, '${address}', '${payment}');`);

    alert('¡Venta procesada con éxito! Registro insertado en la tabla `ventas`.');
    cart = [];
    document.getElementById('cart-count').innerText = '0';
    toggleCart();
}

// Control del Modal de Autenticación
function toggleAuthModal() {
    const modal = document.getElementById('auth-modal');
    modal.style.display = (modal.style.display === 'flex') ? 'none' : 'flex';
}

function switchTab(tab) {
    const loginForm = document.getElementById('form-login');
    const regForm = document.getElementById('form-register');
    const btns = document.querySelectorAll('.tab-btn');

    if (tab === 'login') {
        loginForm.classList.remove('hidden');
        regForm.classList.add('hidden');
        btns[0].classList.add('active');
        btns[1].classList.remove('active');
    } else {
        loginForm.classList.add('hidden');
        regForm.classList.remove('hidden');
        btns[1].classList.add('active');
        btns[0].classList.remove('active');
    }
}

// Log interactivo de SQL en pantalla
function logSQL(query) {
    document.getElementById('sql-log').innerText = query;
}
